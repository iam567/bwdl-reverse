  byte ni = N[exec], ti = N[tgt], p = P[tgt];

void m(short s) {                       // b 7665 重掷忠诚
  int n  = 100 - P[s];
  int n2 = (n + 50) / 2;
  if (n == n2)        n2 = s() % 3 - 1;            // 仅 P==50,招揽路径到不了

}

// 例:P=35 → 57..64;P=0 → 75..99
